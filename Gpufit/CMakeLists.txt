# Native compilation is shared by uv/scikit-build-core and standalone CMake.
option(USE_CUBLAS "Use the cuBLAS solver" ON)
option(DOUBLE_PRECISION "Use the float64 ABI" OFF)
option(BUILD_TESTING "Build native tests" OFF)
if(DOUBLE_PRECISION)
  message(FATAL_ERROR "QI2lab microscopy kernels and Python ABI require DOUBLE_PRECISION=OFF")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "QI2lab GPUfit requires a 64-bit compiler")
endif()
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES "75;80;86;89;90" CACHE STRING "Explicit CUDA architectures")
endif()
enable_language(CUDA)
find_package(CUDAToolkit REQUIRED)
if(NOT CMAKE_CUDA_COMPILER_VERSION VERSION_EQUAL CUDAToolkit_VERSION)
  message(FATAL_ERROR "CUDA compiler and CUDAToolkit versions disagree")
endif()
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)

set(GPUFIT_SOURCE_REVISION "unrecorded-source-archive")
set(GPUFIT_SOURCE_DIRTY false)
find_package(Git QUIET)
if(Git_FOUND AND EXISTS "${PROJECT_SOURCE_DIR}/.git")
  execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
    WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" OUTPUT_VARIABLE GPUFIT_SOURCE_REVISION
    OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain --untracked-files=no
    WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" OUTPUT_VARIABLE source_status
    OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(source_status)
    set(GPUFIT_SOURCE_DIRTY true)
  endif()
elseif(EXISTS "${PROJECT_SOURCE_DIR}/cmake/source_identity.json")
  file(READ "${PROJECT_SOURCE_DIR}/cmake/source_identity.json" source_identity)
  string(JSON GPUFIT_SOURCE_REVISION GET "${source_identity}" source_revision)
  string(JSON archive_dirty GET "${source_identity}" source_dirty)
  if(archive_dirty)
    set(GPUFIT_SOURCE_DIRTY true)
  endif()
endif()
if(SKBUILD_STATE STREQUAL "sdist")
  # Preserve the checkout identity when rebuilding from an archive without .git.
  file(WRITE "${PROJECT_SOURCE_DIR}/cmake/source_identity.json"
    "{\"source_revision\":\"${GPUFIT_SOURCE_REVISION}\",\"source_dirty\":${GPUFIT_SOURCE_DIRTY}}\n")
endif()
if(USE_CUBLAS)
  set(GPUFIT_CUBLAS_JSON true)
else()
  set(GPUFIT_CUBLAS_JSON false)
endif()
configure_file("${PROJECT_SOURCE_DIR}/cmake/build_manifest.json.in"
  "${CMAKE_CURRENT_BINARY_DIR}/build_manifest.json" @ONLY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/build_manifest.json" manifest_text)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gpufit_build_manifest.h"
  "#pragma once\nstatic const char GPUFIT_BUILD_MANIFEST[] = R\"manifest(${manifest_text})manifest\";\n")

file(GLOB model_headers CONFIGURE_DEPENDS "models/*.cuh" "estimators/*.cuh")
add_library(Gpufit SHARED gpufit.cpp info.cpp lm_fit.cpp lm_fit_cuda.cpp interface.cpp
  lm_fit_cuda.cu cuda_kernels.cu info.cu gpu_data.cu cuda_gaussjordan.cu ${model_headers})
if(WIN32)
  target_sources(Gpufit PRIVATE Gpufit.def)
endif()
target_include_directories(Gpufit PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(Gpufit PRIVATE CUDA::cudart_static)
set_target_properties(Gpufit PROPERTIES CUDA_STANDARD 14 CXX_STANDARD 14
  CXX_VISIBILITY_PRESET hidden RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
if(USE_CUBLAS)
  target_compile_definitions(Gpufit PRIVATE USE_CUBLAS)
  target_link_libraries(Gpufit PRIVATE CUDA::cublas)
endif()

install(TARGETS Gpufit RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build_manifest.json" DESTINATION .)
if(USE_CUBLAS AND WIN32)
  if(EXISTS "${CUDAToolkit_LIBRARY_ROOT}/EULA.txt")
    install(FILES "${CUDAToolkit_LIBRARY_ROOT}/EULA.txt" DESTINATION licenses RENAME NVIDIA-CUDA-EULA.txt)
  endif()
  # Query installed files, not toolkit-version-derived DLL names. CUDA 12 cuBLAS
  # depends on cuBLASLt; both are included for clean-machine loading.
  foreach(runtime_name cublas cublasLt)
    file(GLOB runtime_dlls "${CUDAToolkit_BIN_DIR}/${runtime_name}64_*.dll")
    list(LENGTH runtime_dlls runtime_count)
    if(NOT runtime_count EQUAL 1)
      message(FATAL_ERROR "Cannot uniquely locate ${runtime_name} runtime DLL in ${CUDAToolkit_BIN_DIR}")
    endif()
    install(FILES ${runtime_dlls} DESTINATION .)
  endforeach()
elseif(USE_CUBLAS AND UNIX)
  install(IMPORTED_RUNTIME_ARTIFACTS CUDA::cublas CUDA::cublasLt LIBRARY DESTINATION .)
  set_target_properties(Gpufit PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()

if(NOT SKBUILD)
  if(BUILD_TESTING)
    add_subdirectory(tests)
  endif()
  add_subdirectory(examples)
  add_subdirectory(matlab)
  add_subdirectory(java)
  add_subdirectory(python)
endif()
